    <script>
        // 1. Điều khiển Slider 2 Banner tự trượt kích thước bằng nhau
        let currentBannerIndex = 0;
        const totalBanners = 2;
        let bannerInterval = null;

        function showBanner(index) {
            currentBannerIndex = (index + totalBanners) % totalBanners;
            for (let i = 0; i < totalBanners; i++) {
                const slide = document.getElementById('bannerSlide' + i);
                const dot = document.getElementById('bannerDot' + i);
                if (!slide || !dot) continue;
                if (i === currentBannerIndex) {
                    slide.classList.remove('opacity-0', 'invisible', 'z-0');
                    slide.classList.add('opacity-100', 'visible', 'z-10');
                    dot.className = 'w-6 h-1.5 rounded-full bg-duo-cyan transition-all';
                } else {
                    slide.classList.remove('opacity-100', 'visible', 'z-10');
                    slide.classList.add('opacity-0', 'invisible', 'z-0');
                    dot.className = 'w-2.5 h-1.5 rounded-full bg-white/50 transition-all';
                }
            }
        }

        function nextBanner() {
            showBanner(currentBannerIndex + 1);
            resetBannerTimer();
        }

        function prevBanner() {
            showBanner(currentBannerIndex - 1);
            resetBannerTimer();
        }

        function goToBanner(index) {
            showBanner(index);
            resetBannerTimer();
        }

        function resetBannerTimer() {
            if (bannerInterval) clearInterval(bannerInterval);
            bannerInterval = setInterval(() => {
                showBanner(currentBannerIndex + 1);
            }, 5000);
        }

        // 2. Đóng/mở Menu trên điện thoại
        function toggleMobileMenu() {
            const drawer = document.getElementById('mobileMenuDrawer');
            const icon = document.getElementById('mobileMenuIcon');
            if (!drawer) return;
            drawer.classList.toggle('hidden');
            if (drawer.classList.contains('hidden')) {
                icon.className = 'fa-solid fa-bars text-sm';
            } else {
                icon.className = 'fa-solid fa-xmark text-sm';
            }
        }

        // 3. Lọc Danh mục Sản phẩm / Dự án
        function filterCategory(categoryKey) {
            const cards = document.querySelectorAll('.product-card');
            cards.forEach(card => {
                const cardCat = card.getAttribute('data-category');
                if (categoryKey === 'all' || cardCat === categoryKey) {
                    card.classList.remove('hidden');
                } else {
                    card.classList.add('hidden');
                }
            });

            const buttons = document.querySelectorAll('.cat-filter-btn');
            buttons.forEach(btn => {
                if (btn.getAttribute('data-cat-btn') === categoryKey) {
                    btn.className = 'cat-filter-btn cat-single-line px-2.5 py-1.5 rounded-lg text-[11px] sm:text-xs font-semibold bg-duo-dark text-white transition-colors';
                } else {
                    btn.className = 'cat-filter-btn cat-single-line px-2.5 py-1.5 rounded-lg text-[11px] sm:text-xs font-medium bg-slate-100 text-slate-700 hover:bg-slate-200 transition-colors';
                }
            });

            const section = document.getElementById('san-pham-du-an');
            if (section) {
                section.scrollIntoView({ behavior: 'smooth', block: 'start' });
            }
        }

        // 4. Tìm kiếm nhanh sản phẩm
        function searchProductsByKeyword(keyword) {
            const normalized = keyword.trim().toLowerCase();
            const cards = document.querySelectorAll('.product-card');
            cards.forEach(card => {
                const text = card.innerText.toLowerCase();
                if (!normalized || text.includes(normalized)) {
                    card.classList.remove('hidden');
                } else {
                    card.classList.add('hidden');
                }
            });
            const section = document.getElementById('san-pham-du-an');
            if (section) {
                section.scrollIntoView({ behavior: 'smooth', block: 'start' });
            }
        }

        function handleHeaderSearch() {
            const input = document.getElementById('headerSearchInput');
            if (input) searchProductsByKeyword(input.value);
        }

        function handleMobileSearch() {
            const input = document.getElementById('mobileSearchInput');
            if (input) {
                searchProductsByKeyword(input.value);
                toggleMobileMenu();
            }
        }

        // 5. Lọc Danh mục Bài viết trang chủ
        function filterPosts(postCat) {
            const posts = document.querySelectorAll('.blog-card');
            posts.forEach(post => {
                const cat = post.getAttribute('data-post-cat');
                if (postCat === 'all' || cat === postCat) {
                    post.classList.remove('hidden');
                } else {
                    post.classList.add('hidden');
                }
            });
        }

        // 6. Chọn sản phẩm và cuộn xuống Form Báo Giá
        function openQuoteWithProduct(productName) {
            const noteField = document.getElementById('custNote');

        }

        function submitQuoteForm(event) {
            event.preventDefault();
            const name = document.getElementById('custName').value.trim();

                successBox.classList.remove('hidden');
                document.getElementById('quoteForm').reset();
            }
        }

            resetBannerTimer();
